(function() {

const slides = document.querySelectorAll('.hero_slide');

const dots = document.querySelectorAll('.hero_slider-dot');

if (slides.length < 2) return;

let current = 0;

let timer;

function goTo(index, direction) {

if (index === current) return;

var next = (index + slides.length) % slides.length;

var dir = direction || (next > current ? 'right' : 'left');

var exitClass = dir === 'right' ? 'hero_slide--exit-left' : 'hero_slide--exit-right';

var enterClass = dir === 'right' ? 'hero_slide--exit-right' : 'hero_slide--exit-left';

slides[next].classList.add(enterClass);

slides[next].classList.remove('hero_slide--exit-left', 'hero_slide--exit-right');

void slides[next].offsetWidth;

slides[current].classList.remove('hero_slide--active');

slides[current].classList.add(exitClass);

dots[current].classList.remove('hero_slider-dot--active');

slides[next].classList.remove(enterClass);

slides[next].classList.add('hero_slide--active');

dots[next].classList.add('hero_slider-dot--active');

current = next;

}

function startTimer() {

clearInterval(timer);

timer = setInterval(function() { goTo((current + 1) % slides.length, 'right'); }, 5000);

}

dots.forEach(function(dot) {

dot.addEventListener('click', function() {

var idx = parseInt(this.dataset.slide);

goTo(idx, idx > current ? 'right' : 'left');

startTimer();

});

});

startTimer();

})();

</script>
